cmake_minimum_required(VERSION 3.16)
project(ets2_sensor_degradation)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(ets2_msgs REQUIRED)
add_executable(sensor_degradation src/sensor_degradation_node.cpp)
target_compile_features(sensor_degradation PRIVATE cxx_std_20)
target_include_directories(sensor_degradation PRIVATE include)
ament_target_dependencies(sensor_degradation rclcpp sensor_msgs ets2_msgs)
install(TARGETS sensor_degradation DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(models_test test/models_test.cpp)
  target_compile_features(models_test PRIVATE cxx_std_20)
  target_include_directories(models_test PRIVATE include)
endif()
ament_package()
