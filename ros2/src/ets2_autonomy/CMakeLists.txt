cmake_minimum_required(VERSION 3.16)
project(ets2_autonomy)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ets2_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(drive_speed src/gt_drive_node.cpp)
target_compile_features(drive_speed PRIVATE cxx_std_20)
target_link_libraries(drive_speed nlohmann_json::nlohmann_json)
ament_target_dependencies(drive_speed rclcpp ets2_msgs geometry_msgs std_msgs)
add_executable(drive_hold src/drive_hold_node.cpp)
target_compile_features(drive_hold PRIVATE cxx_std_20)
ament_target_dependencies(drive_hold rclcpp ets2_msgs)
install(TARGETS drive_speed drive_hold DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(lateral_sim_test test/lateral_sim_test.cpp)
  target_compile_features(lateral_sim_test PRIVATE cxx_std_20)
  target_include_directories(lateral_sim_test PRIVATE src)
  target_link_libraries(lateral_sim_test nlohmann_json::nlohmann_json)
  # Real map lane from map_lane_provider; the test skips that scenario when absent.
  target_compile_definitions(lateral_sim_test PRIVATE ETS2_LANE_FILE="${CMAKE_CURRENT_SOURCE_DIR}/../../../bridge/recordings/gt-path/current-lane.json")
  ament_add_gtest(longitudinal_test test/longitudinal_test.cpp)
  target_compile_features(longitudinal_test PRIVATE cxx_std_20)
  target_include_directories(longitudinal_test PRIVATE src)
endif()
ament_package()
